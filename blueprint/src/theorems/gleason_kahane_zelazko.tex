\section{Gleason--Kahane--\.{Z}elazko theorem}

\begin{theorem}[Gleason--Kahane--\.{Z}elazko]
\label{thm:gkz}
\lean{GleasonKahaneZelazkoTheorem.MainTheorem}
\leanok
A normalized complex-linear functional on a complex Banach algebra which does not vanish on invertible elements is multiplicative, hence is an algebra homomorphism. The Lean proof derives spectral membership, uses an entire bounded ratio and Liouville's theorem to control exponentials of kernel elements, and then polarizes the square identity.
\end{theorem}
