\section{Burnside's theorem for prime degree}

\begin{theorem}[Burnside prime-degree theorem]
\label{thm:burnside-prime-degree}
\lean{BurnsidePrimeDegreeTheorem.MainTheorem}
\notready
A transitive permutation group of prime degree is either $2$-transitive or has a normal regular subgroup. The formal proof handles the normal Sylow case and several small-prime cases, but the character-theoretic case for primes at least $7$ remains a \texttt{sorry}.
\end{theorem}
