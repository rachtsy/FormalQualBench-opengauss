\section{De Bruijn--Erd\H{o}s theorem}

\begin{theorem}[De Bruijn--Erd\H{o}s]
\label{thm:debruijn-erdos}
\lean{DeBruijnErdos.MainTheorem}
\leanok
If every finite induced subgraph of a graph $G$ is $k$-colorable, then $G$ is $k$-colorable. The proof uses compactness of the product of the finite color spaces and the finite-intersection property.
\end{theorem}
