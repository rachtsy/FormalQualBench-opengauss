\section{Jordan's derangement theorem}

\begin{theorem}[Jordan derangement theorem]
\label{thm:jordan-derangement}
\lean{JordanDerangementTheorem.MainTheorem}
\leanok
Every finite transitive permutation group on a nontrivial set contains a derangement. The proof uses the orbit-counting identity for the sum of the numbers of fixed points and derives a strict counting contradiction if every group element fixes a point.
\end{theorem}
