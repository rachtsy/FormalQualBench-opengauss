\section{Jordan's cycle theorem}

\begin{theorem}[Jordan cycle theorem]
\label{thm:jordan-cycle}
\lean{JordanCycleTheorem.MainTheorem}
\notready
If a primitive permutation group contains a prime cycle whose support is sufficiently small compared with the degree, then it contains the alternating group. The Lean proof handles $p=2,3$ and reduces larger primes to producing a $3$-cycle from $4$-pretransitivity; the final commutator construction remains a \texttt{sorry}.
\end{theorem}
